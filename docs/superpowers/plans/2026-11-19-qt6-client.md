```cpp
#include "net/frame_buffer.h"
#include <QtEndian>
#include <cassert>
#include <iostream>
static QByteArray frame(const QByteArray& p){
    QByteArray f; quint32 n = quint32(p.size());
    f.append(char((n>>24)&0xff)); f.append(char((n>>16)&0xff));
    f.append(char((n>>8)&0xff)); f.append(char(n&0xff)); f.append(p); return f;
}
int main(){
    FrameBuffer fb; QByteArray out;
    fb.feed(frame("{\"a\":1}"));
    assert(fb.next(out) == FrameBuffer::Status::Ok && out == "{\"a\":1}");
    assert(fb.next(out) == FrameBuffer::Status::Incomplete);
    // 半包：先喂 3 字节再喂剩余
    QByteArray f = frame("{\"b\":2}");
    fb.feed(f.left(3)); assert(fb.next(out) == FrameBuffer::Status::Incomplete);
    fb.feed(f.mid(3)); assert(fb.next(out) == FrameBuffer::Status::Ok && out == "{\"b\":2}");
    // 粘包：两帧一次喂
    fb.feed(frame("{\"c\":3}") + frame("{\"d\":4}"));
    assert(fb.next(out) == FrameBuffer::Status::Ok && out == "{\"c\":3}");
    assert(fb.next(out) == FrameBuffer::Status::Ok && out == "{\"d\":4}");
    // 坏头：0xFFFFFFFF → BadHeader 且缓冲清空
    fb.feed(QByteArray::fromHex("ffffffff") + "{\"e\":5}");
    assert(fb.next(out) == FrameBuffer::Status::BadHeader && fb.pending() == 0);
    // 零长度头同样拒
    fb.feed(QByteArray::fromHex("00000000"));
    assert(fb.next(out) == FrameBuffer::Status::BadHeader);
    std::cout << "frame_buffer tests passed\n"; return 0;
}
```

// BubbleDelegate：把一条消息画成圆角气泡；自己发的靠右accent色，对方靠左panel色。
class BubbleDelegate : public QStyledItemDelegate {
public:
    int selfId = 0;
    void paint(QPainter* p, const QStyleOptionViewItem& opt, const QModelIndex& i) const override {
        const bool mine = i.data(Qt::UserRole + 1).toInt() == selfId;
        const QString text = i.data(Qt::DisplayRole).toString();
        QRect r = opt.rect.adjusted(8, 4, -8, -4);
        QFontMetrics fm(opt.font);
        int w = qMin(fm.horizontalAdvance(text) + 24, r.width() * 2 / 3);
        QRect bubble(mine ? r.right() - w : r.left(), r.top(), w, fm.height() + 16);
        p->setRenderHint(QPainter::Antialiasing);
        p->setBrush(mine ? QColor("#E8465A") : QColor("#1B212C"));
        p->setPen(Qt::NoPen);
        p->drawRoundedRect(bubble, 10, 10);
        p->setPen(QColor("#E6E9F0"));
        p->drawText(bubble.adjusted(12, 8, -12, -8), Qt::AlignLeft | Qt::TextWordWrap, text);
    }
    QSize sizeHint(const QStyleOptionViewItem& opt, const QModelIndex& i) const override {
        QFontMetrics fm(opt.font);
        return QSize(opt.rect.width(), fm.height() + 24);
    }
};

#include <QApplication>
#include <QTimer>
#include "app/devmode.h"
#include "app/server_process.h"

    MainWindow w(&sess, &store, &dev);

    sess.connectToServer("127.0.0.1", 12346);
    // --grab <png>：offscreen 目检 hook（QT_QPA_PLATFORM=offscreen 配合）
    for (int i = 1; i < argc; ++i) {
        if (qstrcmp(argv[i], "--grab") == 0 && i + 1 < argc) {
            const QString path = QString::fromLocal8Bit(argv[i + 1]);
            QTimer::singleShot(600, &app, [&]{ w.grab().save(path); app.quit(); });
        }
    }
    const int rc = app.exec();
    server.stop();
    return rc;
}
